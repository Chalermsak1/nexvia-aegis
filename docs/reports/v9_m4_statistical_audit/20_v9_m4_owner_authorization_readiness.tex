\section{Hypotheses H1--H5 Protocol and Implementation Check}

\begin{itemize}
    \item \textbf{H1 (Selective Risk Certification):} Evaluated on mixture holdout $\mathcal{D}_{\text{test}}^{\text{mix}}$ ($N=1,440$). Computes $R_{\text{sel}} = K / M$. Evaluates $M \ge 59$ as eligibility precondition and tests 1-sided exact Clopper--Pearson 95\% UCB $U_{\text{CP}}(K, M; 0.05) \le 0.05$. When $M = 0$, $R_{\text{sel}}$ is evaluated as \texttt{"UNDEFINED"}, safely failing certification without runtime crashes.
    \item \textbf{H2 (False-Success Reduction vs Baselines):} Strictly paired against V8.1 Martingale and V7 MPC using \texttt{align\_and\_pair\_records} over \texttt{(seed, scenario)}. Applies McNemar's exact test and Newcombe's paired-difference confidence interval. Mismatched records fail closed.
    \item \textbf{H3 (TSR Noninferiority):} Evaluated on $S_{10}$ ($N=1,200$ matched pairs) against primary baseline V8.1 and auxiliary baseline V7 with margin $\delta_0 = -0.05$. The confirmatory pass/fail decision is governed strictly by the paired Student's $t$ Wald LCB ($> -0.05$) and nominal TSR $\ge 0.94$. Newcombe Method 10 is reported as a prespecified sensitivity analysis with boolean concurrence.
    \item \textbf{H4 (Coverage Floor Precondition):} Point estimate $\widehat{C} = M / N \ge 0.35$ on $\mathcal{D}_{\text{test}}^{\text{mix}}$ ($M \ge 504$). Anti-gaming logic classifies policies with $\widehat{C} < 0.35$ as \texttt{"CONSERVATIVE\_RISK\_COVERAGE\_TRADEOFF"}, overriding $H_1$.
    \item \textbf{H5 (Worst-Case Scenario Safety):} Evaluates unconditional false-success incidence $R_{\text{inc}, s} = \mathrm{FS}_s / N_s$ across all 12 scenarios in $\mathcal{D}_{\text{test}}^{\text{scen}}$ ($N_s = 100$). Tests global safety via Berger Intersection-Union Test ($\max_s(p_s) \le 0.05$) alongside step-down Holm--Šidák adjusted bounds. Selective risk is undefined under complete scenario abstention ($M_s = 0$).
\end{itemize}

\section{CRN Semantics and Paired Baseline Readiness}
